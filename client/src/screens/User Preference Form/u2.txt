                    name="furnishingLevel"
                    value={formData.furnishingLevel}
                    onChange={handleChange}
                    required
                    style={getSelectStyle(formData.furnishingLevel)}
                    onFocus={(e) => {
                      e.target.style.borderColor = '#22D3EE';
                      e.target.style.background = '#FFFFFF';
                      e.target.style.boxShadow = '0 0 0 3px rgba(34, 211, 238, 0.1)';
                      e.target.style.transform = 'translateY(-2px)';
                    }}
                    onBlur={(e) => {
                      e.target.style = getSelectStyle(formData.furnishingLevel);
                    }}
                  >
                    <option value="">Select Furnishing</option>
                    <option value="fully-furnished">Fully Furnished</option>
                    <option value="semi-furnished">Semi Furnished</option>
                    <option value="unfurnished">Unfurnished</option>
                  </select>
                </div>
              </div>

              <div style={styles.inputField}>
                <label htmlFor="moveInDate" style={styles.label}>Move-in Date</label>
                <div style={{ position: 'relative' }}>
                  <button
                    type="button"
                    onClick={() => setShowDatePicker(!showDatePicker)}
                    style={getDateButtonStyle()}
                    onFocus={(e) => {
                      e.target.style.borderColor = '#22D3EE';
                      e.target.style.background = '#FFFFFF';
                      e.target.style.boxShadow = '0 0 0 3px rgba(34, 211, 238, 0.1)';
                      e.target.style.transform = 'translateY(-2px)';
                    }}
                    onBlur={(e) => {
                      e.target.style = getDateButtonStyle();
                    }}
                  >
                    {formData.moveInDate || 'Select Move-in Date'}
                  </button>
                  
                  {showDatePicker && (
                    <div style={styles.datePickerDropdown}>
                      <button
                        type="button"
                        style={styles.dateOption}
                        onClick={() => handleDateSelect('immediate')}
                        onMouseEnter={(e) => {
                          e.target.style.background = '#22D3EE';
                          e.target.style.color = '#FFFFFF';
                        }}
                        onMouseLeave={(e) => {
                          e.target.style.background = '#F4F7F9';
                          e.target.style.color = '#333333';
                        }}
                      >
                        🚀 Immediate
                      </button>
                      <button
                        type="button"
                        style={styles.dateOption}
                        onClick={() => handleDateSelect('15')}
                        onMouseEnter={(e) => {
                          e.target.style.background = '#22D3EE';
                          e.target.style.color = '#FFFFFF';
                        }}
                        onMouseLeave={(e) => {
                          e.target.style.background = '#F4F7F9';
                          e.target.style.color = '#333333';
                        }}
                      >
                        Within 15 days
                      </button>
                      <button
                        type="button"
                        style={styles.dateOption}
                        onClick={() => handleDateSelect('30')}
                        onMouseEnter={(e) => {
                          e.target.style.background = '#22D3EE';
                          e.target.style.color = '#FFFFFF';
                        }}
                        onMouseLeave={(e) => {
                          e.target.style.background = '#F4F7F9';
                          e.target.style.color = '#333333';
                        }}
                      >
                        Within 30 days
                      </button>
                      <button
                        type="button"
                        style={styles.dateOption}
                        onClick={() => handleDateSelect('flexible')}
                        onMouseEnter={(e) => {
                          e.target.style.background = '#22D3EE';
                          e.target.style.color = '#FFFFFF';
                        }}
                        onMouseLeave={(e) => {
                          e.target.style.background = '#F4F7F9';
                          e.target.style.color = '#333333';
                        }}
                      >
                        📅 Flexible
                      </button>
                      <input
                        type="date"
                        style={styles.customDateInput}
                        onChange={handleCustomDateSelect}
                        onFocus={(e) => {
                          e.target.style.borderColor = '#22D3EE';
                        }}
                      />
                    </div>
                  )}
                </div>
              </div>
            </div>

            {/* Submit Button */}
            <button
              type="submit"
              style={getSubmitButtonStyle()}
              disabled={!isFormValid() || isSubmitting}
              onMouseEnter={(e) => {
                if (isFormValid() && !isSubmitting) {
                  e.target.style.transform = styles.submitBtnHover.transform;
                  e.target.style.boxShadow = styles.submitBtnHover.boxShadow;
                }
              }}
              onMouseLeave={(e) => {
                if (isFormValid() && !isSubmitting) {
                  e.target.style.transform = 'translateY(0)';
                  e.target.style.boxShadow = 'none';
                }
              }}
            >
              {isSubmitting ? (
                <>
                  <div style={styles.spinner}></div>
                  Finding Your Dream Home...
                </>
              ) : (
                'Find My Perfect Home'
              )}
            </button>
          </form>
        </div>
      </div>
    </>
  );
};

export default UserPreferenceForm;
