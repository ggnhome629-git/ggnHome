  async function saveMessageToServer(message) {
    if (!message || !isOnline) return;
    try {
      await fetch(process.env.REACT_APP_SAVE_ASSISTANT_MESSAGE_API, {
        method: "POST",
        credentials: "include",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          sessionId: sessionId || null,
          message,
          timestamp: new Date().toISOString(),
        }),
      });
    } catch (err) {
      console.warn("Save message failed, queueing for retry:", err);
      try {
        const q = JSON.parse(localStorage.getItem(SAVE_QUEUE_KEY) || "[]");
        q.push({ message, timestamp: Date.now() });
        localStorage.setItem(SAVE_QUEUE_KEY, JSON.stringify(q));
      } catch (e) {
        console.warn("Failed to push to local save queue:", e);
      }
    }
  }

                            : "1px solid rgba(74, 106, 138, 0.2)",
                          backdropFilter: 'blur(10px)',
                          maxWidth: '100%',
                          wordWrap: 'break-word',
                        }}>
                          <div style={{ fontWeight: 500, lineHeight: 1.5 }}>
                            {msg.text}
                          </div>
                        </div>
                      </div>
                    </div>
                  ))}
                  {isThinking && <TypingIndicator />}
                  {suggestions.length > 0 && <SuggestionChips />}
                </>
              )}
              <div ref={messagesEndRef} />
            </div>
          )}

          {/* Quick Actions */}
          {sessionId && <QuickActions />}
        </div>

        {/* Modern Bottom Control Panel */}
        {sessionId && (
          <div style={{
            position: "fixed",
            bottom: "20px",
            left: "50%",
            transform: "translateX(-50%)",
            zIndex: 1000,
            display: "flex",
            flexDirection: "column",
            alignItems: "center",
            gap: "20px",
            width: "100%",
            maxWidth: "400px",
            padding: "0 20px",
          }}>
            <ModernMicButton />
          </div>
        )}

        {/* Modern Help Overlay */}
        {showHelp && <ModernHelpOverlay />}
      </div>
    </>
  );
};

export default VoiceAssistantRent;
